  {\large\bfseries Definitions}\par\smallskip
  \begin{minipage}[t]{0.48\linewidth}
    \textbf{Team.} A set $T\subseteq D^V$ of assignments $s:V\to D$.\par
    \smallskip
    \textbf{Dependence atom.} $T\models\mapdep Xy$ means that rows
    agreeing on $X$ also agree on $y$.
  \end{minipage}\hfill
  \begin{minipage}[t]{0.49\linewidth}
    \textbf{Disjunction.} $T\models\bigvee_{i=1}^k\alpha_i$ if
    $T=\bigcup_{i=1}^k T_i$ with $T_i\models\alpha_i$.\par
    \smallskip
    \textbf{Implication.} $\Sigma\models\varphi$ if every team satisfying
    all premises in $\Sigma$ also satisfies $\varphi$.
  \end{minipage}\par\medskip
  {\small\color{mapmuted} $D$ is any nonempty value domain; $V$ contains
  the finitely many input variables. Disjunctions are nonempty.
  Cover parts may overlap or be empty; repeated atoms count separately.}
};
